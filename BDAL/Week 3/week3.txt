Q1) Character Frequency Count: Given a text file, count the frequency of each alphabetic character
(case-insensitive, ignore spaces/punctuation). Hint: modify the mapper to emit one character per
line instead of a word.

#mapper.py
#!/usr/bin/env python3
# mapper.py
import sys

# Read lines from standard input
for line in sys.stdin:
    # Convert to lowercase for case-insensitivity
    line = line.lower()
    
    # Iterate through each character in the line
    for char in line:
        # Emit only if the character is an alphabetic letter (a-z)
        if char.isalpha():
            print(f"{char}\t1")
            
# reducer.py
#!/usr/bin/env python3
# reducer.py
import sys

current_char = None
current_count = 0
char = None

# Read lines from standard input
for line in sys.stdin:
    line = line.strip()
    if not line:
        continue
    
    try:
        char, count = line.split("\t", 1)
        count = int(count)
    except ValueError:
        continue

    # If this is the same character, accumulate its count
    if current_char == char:
        current_count += count
    else:
        if current_char:
            # Emit the previous character and its total count
            print(f"{current_char}\t{current_count}")
        current_char = char
        current_count = count

# Emit the last character
if current_char == char:
    print(f"{current_char}\t{current_count}")

Q3) Maximum Temperature per City: Input: weather.csv with columns City, Date, Temperature.
Find the maximum temperature recorded for each city. Hint: mapper emits city\ttemperature;
reducer tracks the max value seen per key instead of summing.

# mapper.py
#!/usr/bin/env python3
# mapper.py
import sys

# Read lines from standard input
for line in sys.stdin:
    line = line.strip()
    if not line:
        continue
        
    # Skip the header row if present in weather.csv
    if line.startswith("City"):
        continue
        
    # Split the CSV line by commas
    parts = line.split(",")
    
    # Ensure the row has enough columns (City, Date, Temperature)
    if len(parts) >= 3:
        city = parts[0].strip()
        temperature = parts[2].strip()
        
        # Emit city and temperature
        print(f"{city}\t{temperature}")
        
        
        
# reducer.py
#!/usr/bin/env python3
# reducer.py
import sys

current_city = None
max_temp = float('-inf')

# Read lines from standard input
for line in sys.stdin:
    line = line.strip()
    if not line:
        continue
    
    try:
        city, temp_str = line.split("\t", 1)
        temp = float(temp_str)
    except ValueError:
        continue

    # If it's the same city, compare and keep the maximum temperature
    if current_city == city:
        if temp > max_temp:
            max_temp = temp
    else:
        # If we move to a new city, emit the result for the previous city
        if current_city is not None:
            print(f"{current_city}\t{max_temp}")
        current_city = city
        max_temp = temp

# Emit the maximum temperature for the last city
if current_city is not None:
    print(f"{current_city}\t{max_temp}")

Q4) Average Rating per Product: Input: ratings.csv with columns ProductID, UserID, Rating.
Compute the average rating per product. Hint: mapper emits product_id\trating; reducer
accumulates (sum, count) per key and divides at the end.

# mapper.py
#!/usr/bin/env python3
# mapper.py
import sys

# Read lines from standard input
for line in sys.stdin:
    line = line.strip()
    if not line:
        continue
        
    # Skip the header row if present in ratings.csv
    if line.startswith("ProductID"):
        continue
        
    # Split the CSV line by commas
    parts = line.split(",")
    
    # Ensure the row has enough columns (ProductID, UserID, Rating)
    if len(parts) >= 3:
        product_id = parts[0].strip()
        rating = parts[2].strip()
        
        # Emit product_id and rating
        print(f"{product_id}\t{rating}")
        

# reducer.py

#!/usr/bin/env python3
# reducer.py
import sys

current_product = None
total_sum = 0.0
count = 0

# Read lines from standard input
for line in sys.stdin:
    line = line.strip()
    if not line:
        continue
    
    try:
        product_id, rating_str = line.split("\t", 1)
        rating = float(rating_str)
    except ValueError:
        continue

    # If it's the same product, accumulate sum and count
    if current_product == product_id:
        total_sum += rating
        count += 1
    else:
        # If we move to a new product, calculate and emit average for the previous product
        if current_product is not None:
            avg_rating = total_sum / count if count > 0 else 0.0
            print(f"{current_product}\t{avg_rating:.2f}")
        
        current_product = product_id
        total_sum = rating
        count = 1

# Emit the average rating for the last product
if current_product is not None:
    avg_rating = total_sum / count if count > 0 else 0.0
    print(f"{current_product}\t{avg_rating:.2f}")
