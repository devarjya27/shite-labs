# mapper_add_sub.py
import sys
for line in sys.stdin:
    m, r, c, v = line.strip().split(',')
    for k in range(3):
        if m == 'a': print(f"{r},{k}\ta\t{c}\t{v}")
        else: print(f"{k},{c}\tb\t{r}\t{v}")
    
    
# reducer_add_sub.py
import sys
d = {}
for line in sys.stdin:
    k, m, v = line.strip().split('\t')
    d[m] = int(v)
    if 'a' in d and 'b' in d:
        print(f"{k} -> Add: {d['a']+d['b']} | Sub: {d['a']-d['b']}")
        d.clear()

# mapper_mul.py
import sys
for line in sys.stdin:
    m, r, c, v = line.strip().split(',')
    for k in range(3):
        if m == 'a': print(f"{r},{k}\ta\t{c}\t{v}")
        else: print(f"{k},{c}\tb\t{r}\t{v}")
        

# reducer_mul.py
import sys
cur, A, B = None, {}, {}
for line in sys.stdin:
    k, m, i, v = line.strip().split('\t')
    if cur and cur != k:
        ans = sum(A.get(i,0) * B.get(i,0) for i in range(3))
        print(f"{cur} = {ans}")
        A, B = {}, {}
    cur = k
    (A if m == 'a' else B)[int(i)] = int(v)
if cur:
    ans = sum(A.get(i,0) * B.get(i,0) for i in range(3))
    print(f"{cur} = {ans}")

# mapper_sort.py
import sys
for line in sys.stdin:
    line = line.strip()
    if not line:
        continue
    reg_no, name, marks = line.split(',')
    # Output Name as Key, and (RegNo, Marks) as Value
    print(f"{name}\t{reg_no},{marks}")
    
# reducer_sort.py
import sys
for line in sys.stdin:
    name, val = line.strip().split('\t')
    reg_no, marks = val.split(',')
    # Format the final sorted output: RegNo, Name, Marks
    print(f"{reg_no},{name},{marks}")
