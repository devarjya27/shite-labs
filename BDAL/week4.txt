#!/usr/bin/env python3
# reducer.py
import sys

current_word = None
current_count = 0
word = None

# Read lines from standard input
for line in sys.stdin:
    # Remove leading/trailing whitespace and split by tab
    line = line.strip()
    if not line:
        continue
    
    try:
        word, count = line.split("\t", 1)
        count = int(count)
    except ValueError:
        continue

    # If this is the same word, accumulate count
    if current_word == word:
        current_count += count
    else:
        if current_word:
            # Emit the previous word and its count
            print(f"{current_word}\t{current_count}")
        current_word = word
        current_count = count

# Emit the last word
if current_word == word:
    print(f"{current_word}\t{current_count}")

#!/usr/bin/env python3
import sys

def main():
    current_cust_id = None
    customer_info = []
    orders_list = []

    for line in sys.stdin:
        line = line.strip()
        if not line:
            continue
        
        cust_id, value = line.split('\t', 1)
        tag, rest = value.split(',', 1)

        if current_cust_id != cust_id:
            if current_cust_id is not None:
                for c_name, c_city in customer_info:
                    for o_id, o_amount in orders_list:
                        print(f"{current_cust_id}\t{c_name}\t{c_city}\t{o_id}\t{o_amount}")
            
            current_cust_id = cust_id
            customer_info = []
            orders_list = []

        if tag == "CUST":
            name, city = rest.split(',', 1)
            customer_info.append((name, city))
        elif tag == "ORD":
            order_id, amount = rest.split(',', 1)
            orders_list.append((order_id, amount))

    if current_cust_id is not None:
        for c_name, c_city in customer_info:
            for o_id, o_amount in orders_list:
                print(f"{current_cust_id}\t{c_name}\t{c_city}\t{o_id}\t{o_amount}")

if __name__ == "__main__":
    main()

#!/usr/bin/env python3
import sys

def main():
    word_counts = []

    for line in sys.stdin:
        line = line.strip()
        if not line:
            continue
        parts = line.split('\t', 1)
        if len(parts) != 2:
            continue
        
        rest = parts[1]
        word, count_str = rest.rsplit(',', 1)
        try:
            count = int(count_str)
        except ValueError:
            continue
        
        word_counts.append((count, word))

    word_counts.sort(key=lambda x: x[0], reverse=True)
    top_10 = word_counts[:10]

    for count, word in top_10:
        print(f"{word}\t{count}")

if __name__ == "__main__":
    main()

#!/usr/bin/env python3
import sys

def main():
    window_size = 2
    for line in sys.stdin:
        line = line.strip()
        if not line:
            continue
        words = line.split()
        n = len(words)
        for i in range(n):
            for j in range(i + 1, min(i + 1 + window_size, n)):
                w1 = words[i]
                w2 = words[j]
                if w1 != w2:
                    pair = sorted([w1, w2])
                    print(f"{pair[0]},{pair[1]}\t1")

if __name__ == "__main__":
    main()

#!/usr/bin/env python3
import sys

def main():
    current_pair = None
    current_count = 0

    for line in sys.stdin:
        line = line.strip()
        if not line:
            continue
        parts = line.split('\t', 1)
        if len(parts) != 2:
            continue
        pair, count_str = parts
        try:
            count = int(count_str)
        except ValueError:
            continue

        if current_pair == pair:
            current_count += count
        else:
            if current_pair is not None:
                print(f"{current_pair}\t{current_count}")
            current_pair = pair
            current_count = count

    if current_pair is not None:
        print(f"{current_pair}\t{current_count}")

if __name__ == "__main__":
    main()
